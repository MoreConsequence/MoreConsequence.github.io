## 可复用导出片段（diagram-design HTML→SVG）

```bash
cd /Users/lianghaoyu/codes/github-blog
SKILL=~/.codex/skills/diagram-design
SRC=diagrams/<dir>/<name>.html
OUT=public/images/<name>.svg
python3 - "$SRC" "$OUT" <<'PY'
import sys, re, pathlib
src, out = sys.argv[1], sys.argv[2]
html = pathlib.Path(src).read_text(encoding="utf-8")
m = re.search(r"<svg\b.*?</svg>", html, re.S)
svg = m.group(0)
if "xmlns=" not in svg.split(">",1)[0]:
    svg = svg.replace("<svg", '<svg xmlns="http://www.w3.org/2000/svg"', 1)
svg = '<?xml version="1.0" encoding="UTF-8"?>\n' + svg
pathlib.Path(out).write_text(svg, encoding="utf-8")
PY
python3 "$SKILL/scripts/self_check.py" "$SRC"
```
